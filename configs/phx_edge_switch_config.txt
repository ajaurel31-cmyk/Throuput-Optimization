! PHX Edge Switch - Cisco Catalyst 9300
! WAN-facing switch on the Phoenix side

hostname PHX-EDGE-SW01

spanning-tree mode rapid-pvst

! WAN uplink (10G circuit from ATL)
interface TenGigabitEthernet1/0/48
 description TO-WAN-10G-ATL-PHX
 no switchport
 ip address 172.16.0.2 255.255.255.252
 ip mtu 1500
 ip tcp adjust-mss 1360
 speed 10000
 duplex full
 service-policy output WAN-QOS
 no shutdown

! Downlink to firewall
interface TenGigabitEthernet1/0/1
 description TO-PHX-FW01-eth1/2
 switchport mode trunk
 switchport trunk allowed vlan 210,999
 speed 10000
 duplex full
 no shutdown

! WAN QoS
policy-map WAN-QOS
 class PRIORITY
  priority percent 10
 class BULK-TRANSFER
  bandwidth remaining percent 50
  shape average 5000000000 bps
 class class-default
  bandwidth remaining percent 40

! Routing
router ospf 1
 network 172.16.0.0 0.0.0.3 area 0
 network 10.2.210.0 0.0.0.3 area 0

ip route 10.1.0.0 255.255.0.0 172.16.0.1
